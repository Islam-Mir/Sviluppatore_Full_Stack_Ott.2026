CREATE DATABASE IF NOT EXISTS agenzia_viaggi;
USE agenzia_viaggi;



CREATE TABLE clienti (
  id INT AUTO_INCREMENT PRIMARY KEY,
  nome VARCHAR(100) NOT NULL,
  cognome VARCHAR(100) NOT NULL,
  email VARCHAR(150),
  telefono VARCHAR(50),
  nazione VARCHAR(100)
);




CREATE TABLE destinazioni (
  id INT AUTO_INCREMENT PRIMARY KEY,
  citta VARCHAR(100) NOT NULL,
  paese VARCHAR(100) NOT NULL,
  prezzo DECIMAL(10,2) NOT NULL,
  data_partenza DATE NOT NULL,
  data_ritorno DATE NOT NULL,
  posti_disponiibli INT NOT NULL
);



CREATE TABLE prenotazioni (
  id INT AUTO_INCREMENT PRIMARY KEY,
  id_cliente INT NOT NULL,
  id_destinazione INT NOT NULL,
  data_prenotazione DATE NOT NULL,
  num_persone INT NOT NULL,

  CONSTRAINT fk_cliente FOREIGN KEY (id_cliente)    
    REFERENCES clienti(id)
    ON DELETE CASCADE,                          

  CONSTRAINT fk_destinazione FOREIGN KEY (id_destinazione)
    REFERENCES destinazioni(id)
    ON DELETE CASCADE
);



I clienti e le destinazioni sono entità indipendenti.

Le prenotazioni dipendono da entrambe tramite chiavi esterne.

I vincoli ON DELETE CASCADE fanno sì che, eliminando un cliente o una destinazione, vengano cancellate automaticamente anche le prenotazioni collegate.
